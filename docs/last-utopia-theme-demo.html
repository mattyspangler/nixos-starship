# Example Python function with Last Utopia syntax highlighting
import json
from typing import Dict, List

def process_data(data: Dict) -> List:
    """Process input data and return results."""
    results = []
    
    for item in data.get("items", []):
        if item["type"] == "primary":
            results.append({
                "name": item["name"],
                "value": item["value"] * 2,
                "status": "processed"
            })
        elif item["type"] == "secondary":
            results.append({
                "name": item["name"],
                "value": item["value"] + 10,
                "status": "enhanced"
            })
    
    return results

# Example usage
sample_data = {
    "items": [
        {"name": "Primary Item", "type": "primary", "value": 42},
        {"name": "Secondary Item", "type": "secondary", "value": 15}
    ]
}

processed = process_data(sample_data)
print(f"Processed {len(processed)} items")
